# Tareas programadas de Tornea - Terobyte
# Para los scripts copiados sueltos a /home/terobyte/scripts.
# Va en el crontab de ROOT (los scripts necesitan root):
#   sudo crontab -l                       <- ver si root ya tiene tareas
#   sudo crontab crontab_root.txt         <- instalar (REEMPLAZA las que haya)
#
# minuto hora día mes día-semana comando

# Respaldo Tornea - todos los días a las 02:00
0 2 * * * /home/terobyte/scripts/backup_tornea.sh

# Monitoreo Tornea - cada 30 minutos
*/30 * * * * /home/terobyte/scripts/monitoreo_tornea.sh

# Informe de la base de datos - todos los días a las 06:00
0 6 * * * /home/terobyte/scripts/monitoreo_bd.sh --informe

# Antivirus - domingos a las 03:00 (después del respaldo)
0 3 * * 0 /home/terobyte/scripts/seguridad_tornea.sh --escaneo

# Al prender el servidor, vuelve a bloquear las IPs de la lista
# (se espera un minuto a que Docker cree la cadena DOCKER-USER)
@reboot sleep 60 && /home/terobyte/scripts/seguridad_tornea.sh --reaplicar
